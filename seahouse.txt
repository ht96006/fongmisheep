        if ($decoded && strpos($decoded, 'http') === 0) {
            // 取出 ips 參數當作真實 IP
            if (preg_match('/ips=([^&]+)/', $decoded, $ip_match)) {
                $real_ip = urldecode($ip_match[1]);
                // 把 127.0.0.1:9560 換成真實 IP
                $real_url = str_replace('127.0.0.1:9560', $real_ip, $decoded);
                // 清除多餘參數
                $real_url = preg_replace('/[?&]v\.channelid=\d+/', '', $real_url);
                $real_url = preg_replace('/[?&]v\.type=\d+/', '', $real_url);
                $real_url = preg_replace('/[?&]ips=[^&]+/', '', $real_url);
                $real_url = rtrim($real_url, '?&');
            } else {
                $real_url = $decoded;
            }
        }
    }

    echo "#EXTINF:-1 group-title=\"{$current_group}\",{$name}\n";
    echo "{$real_url}\n";
}
